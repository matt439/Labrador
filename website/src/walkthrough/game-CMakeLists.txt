# The game's rules, in a library that links the engine's maths and nothing else
# of it, so the tests below run them with no window and no device.
add_library(MyGameRules STATIC
    rules/motion.cpp
)
target_link_libraries(MyGameRules PUBLIC
    MattMath
    labrador_settings
)

# Includes are written from the project root - "game/states/hello_state.h" -
# so the directory above game/ is the include root, for the rules and for
# everything that links them.
target_include_directories(MyGameRules PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")

# The rules' tests, with doctest from vcpkg. A new test file is listed here.
find_package(doctest CONFIG REQUIRED)
add_executable(MyGameTests
    tests/motion_tests.cpp
)
target_link_libraries(MyGameTests PRIVATE
    MyGameRules
    doctest::doctest
    labrador_settings
)
add_test(NAME MyGameTests COMMAND MyGameTests)

# The browser build: main_web.cpp in place of main.cpp, a page made from
# web/shell.html, and the content packed into the module at the paths the
# Windows build copies it to. Under a wasm preset this is the whole build, and
# the Windows half below is never read.
if(EMSCRIPTEN)
    # The rules' tests run under Node, which ctest starts for them.
    target_link_options(MyGameTests PRIVATE -sENVIRONMENT=node)

    add_executable(MyGame
        main_web.cpp
        states/confirm_state.cpp
        states/hello_state.cpp
    )
    target_link_libraries(MyGame PRIVATE
        MyGameRules
        LabradorEngine
        labrador_settings
    )
    set_target_properties(MyGame PROPERTIES SUFFIX ".html")
    # Neither the page nor the content is a source, so name them here, or a
    # change to either does not relink.
    set_property(TARGET MyGame APPEND PROPERTY LINK_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html"
        "${CMAKE_CURRENT_SOURCE_DIR}/content/manifest.json"
        "${CMAKE_CURRENT_SOURCE_DIR}/content/fonts/jetbrains_mono_bold_16.spritefont"
    )
    target_link_options(MyGame PRIVATE
        "SHELL:--embed-file ${CMAKE_CURRENT_SOURCE_DIR}/content@/"
        "SHELL:--shell-file ${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html"
        -sENVIRONMENT=web
    )
    return()
endif()

add_executable(MyGame WIN32
    main.cpp
    states/confirm_state.cpp
    states/hello_state.cpp
)
target_link_libraries(MyGame PRIVATE
    MyGameRules
    LabradorEngine
    labrador_settings
)

# The start-up check: the game hidden, a few frames drawn, and out. It needs a
# graphics device, and a missing content file fails it naming the file.
add_test(NAME MyGameSmokeTests COMMAND MyGame --smoke-test)
set_tests_properties(MyGameSmokeTests PROPERTIES TIMEOUT 60)

# Content is found beside the executable, so copy it there on every build.
add_custom_target(game_content ALL)
add_custom_command(TARGET game_content POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/content/manifest.json"
        "$<TARGET_FILE_DIR:MyGame>/manifest.json"
    VERBATIM
)
add_custom_command(TARGET game_content POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/content/fonts"
        "$<TARGET_FILE_DIR:MyGame>/fonts"
    VERBATIM
)
add_dependencies(MyGame game_content)
